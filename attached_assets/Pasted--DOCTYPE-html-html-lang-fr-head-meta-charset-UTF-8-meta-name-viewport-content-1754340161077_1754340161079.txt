        let currentFilter = 'all';
        const timelineEl = document.getElementById('timeline');
        const yearMarkersEl = document.getElementById('year-markers');

        function getTypeClass(type) {
            return `type-${type.replace(/[\s-]/g, '-').toLowerCase()}`;
        }

        function getTypeName(type) {
            const typeNames = {
                'dos': 'DOS',
                'theft': 'Theft',
                'fake-conf': 'Fake Conf',
                'netset': 'Netset',
                'exposure': 'Exposure',
                'unknown': 'Unknown',
                'inflation': 'Inflation'
            };
            return typeNames[type] || type;
        }

        function createTimeline() {
            const minYear = Math.min(...vulnerabilities.map(v => v.year));
            const maxYear = Math.max(...vulnerabilities.map(v => v.year));
            const yearSpan = maxYear - minYear;

            // Créer les marqueurs d'années
            yearMarkersEl.innerHTML = '';
            for (let year = minYear; year <= maxYear; year++) {
                const marker = document.createElement('div');
                marker.className = 'year-marker';
                marker.style.left = `${((year - minYear) / yearSpan) * 100}%`;
                
                marker.innerHTML = `
                    <div class="year-dot"></div>
                    <div class="year-label">${year}</div>
                `;
                yearMarkersEl.appendChild(marker);
            }

            // Créer les vulnérabilités
            timelineEl.querySelectorAll('.vulnerability').forEach(el => el.remove());
            
            vulnerabilities.forEach((vuln, index) => {
                const vulnEl = document.createElement('div');
                vulnEl.className = `vulnerability ${getTypeClass(vuln.type)} ${index % 2 === 0 ? 'top' : 'bottom'}`;
                vulnEl.dataset.type = vuln.type;
                
                const position = ((vuln.year - minYear + (vuln.month || 6) / 12) / yearSpan) * 100;
                vulnEl.style.left = `${Math.min(Math.max(position, 5), 85)}%`;
                vulnEl.style.animationDelay = `${index * 0.1}s`;

                vulnEl.innerHTML = `
                    <div class="vuln-header">${vuln.title}</div>
                    <div class="vuln-cve">${vuln.cve}</div>
                    <div class="vuln-type">${getTypeName(vuln.type)}</div>
                    <div class="vuln-description">${vuln.description}</div>
                `;

                vulnEl.addEventListener('click', () => {
                    if (vuln.cve.startsWith('CVE-')) {
                        window.open(`https://cve.mitre.org/cgi-bin/cvename.cgi?name=${vuln.cve}`, '_blank');
                    }
                });

                timelineEl.appendChild(vulnEl);
            });

            updateStats();
        }

        function filterVulnerabilities(type) {
            currentFilter = type;
            const vulnElements = document.querySelectorAll('.vulnerability');
            
            vulnElements.forEach(el => {
                if (type === 'all' || el.dataset.type === type) {
                    el.style.display = 'block';
                } else {
                    el.style.display = 'none';
                }
            });

            updateStats();
        }

        function updateStats() {
            const visibleVulns = vulnerabilities.filter(v => 
                currentFilter === 'all' || v.type === currentFilter
            );

            const criticalTypes = ['inflation', 'theft'];
            const criticalCount = visibleVulns.filter(v => criticalTypes.includes(v.type)).length;
            
            const years = [...new Set(visibleVulns.map(v => v.year))];

            document.getElementById('total-vulns').textContent = visibleVulns.length;
            document.getElementById('critical-vulns').textContent = criticalCount;
            document.getElementById('years-span').textContent = years.length;
        }

        // Event listeners pour les filtres
        document.querySelectorAll('.filter-btn').forEach(btn => {
            btn.addEventListener('click', () => {
                document.querySelectorAll('.filter-btn').forEach(b => b.classList.remove('active'));
                btn.classList.add('active');
                filterVulnerabilities(btn.dataset.type);
            });
        });

        // Initialisation
        createTimeline();
